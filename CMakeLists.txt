cmake_minimum_required(VERSION 3.22.1)
project(CorvusRetro LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CORVUS_ROOT "${CMAKE_CURRENT_LIST_DIR}")
set(BUILD_TESTING OFF CACHE BOOL "Tests are built separately" FORCE)
set(CORVUSCORE_BUILD_CORVUSAPI ON CACHE BOOL "Build bundled CorvusAPI modules" FORCE)
set(CORVUSCORE_INSTALL OFF CACHE BOOL "Embedded dependency" FORCE)
add_subdirectory("${CORVUS_ROOT}/deps/CorvusCore" corvuscore)
# Debug APKs still need optimized emulation; keep debug symbols and assertions.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(core_target corvuscore_common gbc_hardware gb_corvus_core gbc_corvus_core gba_corvus_core corvuscore_c_api)
        get_target_property(core_type ${core_target} TYPE)
        if(NOT core_type STREQUAL "INTERFACE_LIBRARY")
            target_compile_options(${core_target} PRIVATE -O2)
        endif()
    endforeach()
endif()
# Community cores are selected by the application distributor at build time.
# The list becomes a generated native catalog, not a frontend source-code list.
set(CORVUS_EXTRA_CORE_DIRECTORIES "" CACHE STRING
    "Semicolon-separated additional core source directories, absolute or relative to repository root")
set(CORVUS_EXTRA_CORE_TARGETS "" CACHE STRING
    "Semicolon-separated native shared library targets provided by those directories")
# The base APK contains NO emulators. Native core builders explicitly
# enable one or several official engines with -D CORVUS_OFFICIAL_CORES=...
# Use comma-separated IDs so Gradle can pass it safely as one CMake argument.
set(CORVUS_OFFICIAL_CORES "" CACHE STRING
    "Comma-separated official cores bundled in an APK pack: gb,gbc,gba")
string(REPLACE "," ";" _corvus_official_list "${CORVUS_OFFICIAL_CORES}")
set(CORVUS_BUNDLED_CORE_TARGETS "")
foreach(core IN LISTS _corvus_official_list)
    if(core STREQUAL "")
        continue()
    endif()
    if(NOT core MATCHES "^(gb|gbc|gba)$")
        message(FATAL_ERROR "Unknown official CorvusCore core: ${core}")
    endif()
    if("corvus_${core}" IN_LIST CORVUS_BUNDLED_CORE_TARGETS)
        message(FATAL_ERROR "Duplicate bundled CorvusCore core: ${core}")
    endif()
    list(APPEND CORVUS_BUNDLED_CORE_TARGETS corvus_${core})
endforeach()
set(_corvus_extra_index 0)
foreach(extra_dir IN LISTS CORVUS_EXTRA_CORE_DIRECTORIES)
    get_filename_component(extra_source "${extra_dir}" ABSOLUTE BASE_DIR "${CORVUS_ROOT}")
    if(NOT EXISTS "${extra_source}/CMakeLists.txt")
        message(FATAL_ERROR "Community core directory has no CMakeLists.txt: ${extra_source}")
    endif()
    add_subdirectory("${extra_source}" "${CMAKE_CURRENT_BINARY_DIR}/community-${_corvus_extra_index}")
    math(EXPR _corvus_extra_index "${_corvus_extra_index} + 1")
endforeach()
list(APPEND CORVUS_BUNDLED_CORE_TARGETS ${CORVUS_EXTRA_CORE_TARGETS})
list(LENGTH CORVUS_BUNDLED_CORE_TARGETS _corvus_bundled_count)
if(_corvus_bundled_count GREATER 32)
    message(FATAL_ERROR "CorvusRetro supports at most 32 bundled cores")
endif()

set(_corvus_catalog "// CorvusRetro - CorvusSoftware\n// Copyright (C) 2026 CorvusSoftware\n// SPDX-License-Identifier: AGPL-3.0-only\nstatic const char *const cr_bundled_core_modules[] = {\n    0,\n")
foreach(core_target IN LISTS CORVUS_BUNDLED_CORE_TARGETS)
    if(NOT TARGET ${core_target})
        message(FATAL_ERROR "CorvusAPI core target missing: ${core_target}")
    endif()
    get_target_property(_corvus_target_type ${core_target} TYPE)
    if(NOT _corvus_target_type STREQUAL "SHARED_LIBRARY")
        message(FATAL_ERROR "CorvusAPI core must be a shared library: ${core_target}")
    endif()
    string(APPEND _corvus_catalog "    \"$<TARGET_FILE_NAME:${core_target}>\",\n")
endforeach()
string(APPEND _corvus_catalog "};\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/cr_bundled_cores.h"
    CONTENT "${_corvus_catalog}")

option(CORVUS_SANITIZE "Instrument the original portable C code" OFF)
if(ANDROID)
    set(CORVUS_CORE_OUTPUT_DIR "${CMAKE_BINARY_DIR}/native-cores" CACHE PATH "Native archive build outputs")
    foreach(core gb gbc gba)
        set_target_properties(corvus_${core} PROPERTIES LIBRARY_OUTPUT_DIRECTORY
            "${CORVUS_CORE_OUTPUT_DIR}/${CMAKE_BUILD_TYPE}/${ANDROID_ABI}")
        target_link_options(corvus_${core} PRIVATE "-Wl,-z,max-page-size=16384")
    endforeach()
endif()
set(CORVUS_INCLUDE_DIRS "${CORVUS_ROOT}/menu" "${CORVUS_ROOT}/cores"
    "${CORVUS_ROOT}/frontend" "${CORVUS_ROOT}/frontend/drivers"
    "${CORVUS_ROOT}/gfx" "${CORVUS_ROOT}/gfx/drivers" "${CORVUS_ROOT}/input" "${CORVUS_ROOT}/audio/drivers" "${CORVUS_ROOT}/tasks")
include(cmake/Rcheevos.cmake)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(corvus_frontend STATIC cores/engine.c cores/core_registry.c menu/menu_driver.c
    menu/theme.c gfx/viewport.c input/play_layout.c)
target_link_libraries(corvus_frontend PUBLIC CorvusAPI::CorvusAPI corvus_cheevos ${CMAKE_DL_LIBS} m)
if(CORVUS_BUNDLED_CORE_TARGETS)
    add_dependencies(corvus_frontend ${CORVUS_BUNDLED_CORE_TARGETS})
endif()
target_include_directories(corvus_frontend PUBLIC ${CORVUS_INCLUDE_DIRS} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(corvus_frontend PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
    target_include_directories(native_app_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
    add_library(corvusretro SHARED frontend/drivers/platform_android.c frontend/drivers/android_backend.c
        audio/drivers/aaudio.c gfx/drivers/gl2.c)
    target_sources(corvusretro PRIVATE tasks/task_achievements.c)
    target_link_libraries(corvusretro PRIVATE corvus_frontend native_app_glue android EGL GLESv2 aaudio jnigraphics log)
    target_compile_options(corvusretro PRIVATE -Wall -Wextra -Werror)
    target_link_options(corvusretro PRIVATE "-uANativeActivity_onCreate" "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    add_executable(native_engine_test "${CORVUS_ROOT}/tests/native_engine_test.c")
    target_link_libraries(native_engine_test PRIVATE corvus_frontend)
    target_compile_options(native_engine_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_engine COMMAND native_engine_test)
    add_executable(native_frontend_test "${CORVUS_ROOT}/tests/native_frontend_test.c")
    target_link_libraries(native_frontend_test PRIVATE corvus_frontend)
    target_compile_options(native_frontend_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_frontend COMMAND native_frontend_test)
    add_executable(native_viewport_test "${CORVUS_ROOT}/tests/native_viewport_test.c")
    target_link_libraries(native_viewport_test PRIVATE corvus_frontend)
    target_compile_options(native_viewport_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_viewport COMMAND native_viewport_test)
    add_executable(native_controls_test "${CORVUS_ROOT}/tests/native_controls_test.c")
    target_link_libraries(native_controls_test PRIVATE corvus_frontend)
    target_compile_options(native_controls_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_controls COMMAND native_controls_test)
    add_executable(native_menu_test "${CORVUS_ROOT}/tests/native_menu_test.c")
    target_link_libraries(native_menu_test PRIVATE corvus_frontend)
    target_compile_options(native_menu_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_menu COMMAND native_menu_test)
    # A separate empty catalog models a base APK whose installed features arrive
    # in a different order from the three historic console storage suffixes.
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-empty")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/test-empty/cr_bundled_cores.h"
        "static const char *const cr_bundled_core_modules[] = {0};\n")
    add_executable(native_registry_test "${CORVUS_ROOT}/tests/native_registry_test.c"
        cores/engine.c cores/core_registry.c menu/menu_driver.c menu/theme.c gfx/viewport.c input/play_layout.c)
    target_include_directories(native_registry_test PRIVATE
        ${CORVUS_INCLUDE_DIRS} "${CMAKE_CURRENT_BINARY_DIR}/test-empty")
    target_link_libraries(native_registry_test PRIVATE CorvusAPI::CorvusAPI corvus_cheevos ${CMAKE_DL_LIBS} m)
    target_compile_options(native_registry_test PRIVATE -Wall -Wextra -Werror)
    add_dependencies(native_registry_test corvus_gb corvus_gbc corvus_gba)
    add_test(NAME native_registry COMMAND native_registry_test
        "$<TARGET_FILE:corvus_gba>" "$<TARGET_FILE:corvus_gb>" "$<TARGET_FILE:corvus_gbc>")
    if(NOT CORVUS_BUNDLED_CORE_TARGETS)
        add_executable(native_nocore_test "${CORVUS_ROOT}/tests/native_nocore_test.c")
        target_link_libraries(native_nocore_test PRIVATE corvus_frontend)
        target_compile_options(native_nocore_test PRIVATE -Wall -Wextra -Werror)
        add_test(NAME native_nocore COMMAND native_nocore_test)
    endif()
    add_executable(native_cheevos_test tests/native_cheevos_test.c)
    target_link_libraries(native_cheevos_test PRIVATE corvus_cheevos)
    target_compile_options(native_cheevos_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_cheevos COMMAND native_cheevos_test)
    add_executable(native_achievement_tasks_test tests/native_achievement_tasks_test.c tasks/task_achievements.c)
    target_include_directories(native_achievement_tasks_test PRIVATE "${CORVUS_ROOT}/tasks")
    target_link_libraries(native_achievement_tasks_test PRIVATE Threads::Threads)
    target_compile_options(native_achievement_tasks_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_achievement_tasks COMMAND native_achievement_tasks_test)
    if(CORVUS_SANITIZE)
        foreach(target corvus_frontend corvus_cheevos native_cheevos_test native_achievement_tasks_test native_engine_test native_frontend_test native_viewport_test native_controls_test native_menu_test native_registry_test)
            target_compile_options(${target} PRIVATE -g -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
        endforeach()
        foreach(target native_cheevos_test native_achievement_tasks_test native_engine_test native_frontend_test native_viewport_test native_controls_test native_menu_test native_registry_test)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endforeach()
    endif()
endif()
