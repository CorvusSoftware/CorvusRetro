cmake_minimum_required(VERSION 3.22.1)
project(CorvusRetroCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "Core tests run in the frontend host build" FORCE)
set(CORVUSCORE_BUILD_CORVUSAPI ON CACHE BOOL "Use the public CorvusAPI adapter" FORCE)
set(CORVUSCORE_BUILD_LIBRETRO OFF CACHE BOOL "No libretro adapters in Android splits" FORCE)
set(CORVUSCORE_INSTALL OFF CACHE BOOL "No dependency installation" FORCE)
if(NOT CORVUS_FEATURE_CORE MATCHES "^(gb|gbc|gba)$")
    message(FATAL_ERROR "Select one official core: gb, gbc or gba")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../third_party/CorvusCore" corvuscore)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(core_target corvuscore_common gbc_hardware gb_corvus_core gbc_corvus_core gba_corvus_core corvuscore_c_api)
        get_target_property(core_type ${core_target} TYPE)
        if(NOT core_type STREQUAL "INTERFACE_LIBRARY")
            target_compile_options(${core_target} PRIVATE -O2)
        endif()
    endforeach()
endif()
target_link_options(corvus_${CORVUS_FEATURE_CORE} PRIVATE "-Wl,-z,max-page-size=16384")
