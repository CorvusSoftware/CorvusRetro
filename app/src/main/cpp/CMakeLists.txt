cmake_minimum_required(VERSION 3.22.1)
project(CorvusRetro LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CORVUS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../..")
set(BUILD_TESTING OFF CACHE BOOL "Tests are built separately" FORCE)
set(CORVUSCORE_BUILD_LIBRETRO OFF CACHE BOOL "Use the native C API" FORCE)
set(CORVUSCORE_INSTALL OFF CACHE BOOL "Embedded dependency" FORCE)
add_subdirectory("${CORVUS_ROOT}/third_party/CorvusCore" corvuscore)
# Debug APKs still need optimized emulation; keep debug symbols and assertions.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(core_target corvuscore_common gbc_hardware gb_corvus_core gbc_corvus_core gba_corvus_core corvuscore_c_api)
        get_target_property(core_type ${core_target} TYPE)
        if(NOT core_type STREQUAL "INTERFACE_LIBRARY")
            target_compile_options(${core_target} PRIVATE -O2)
        endif()
    endforeach()
endif()
option(CORVUS_SANITIZE "Instrument the original portable C code" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(corvus_frontend STATIC engine.c frontend.c theme.c viewport.c)
target_link_libraries(corvus_frontend PUBLIC corvuscore_c_api m)
target_include_directories(corvus_frontend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(corvus_frontend PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
    target_include_directories(native_app_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
    add_library(corvusretro SHARED native_main.c android_backend.c audio_android.c)
    target_link_libraries(corvusretro PRIVATE corvus_frontend native_app_glue android EGL GLESv2 aaudio jnigraphics log)
    target_compile_options(corvusretro PRIVATE -Wall -Wextra -Werror)
    target_link_options(corvusretro PRIVATE "-uANativeActivity_onCreate" "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    add_executable(native_engine_test "${CORVUS_ROOT}/tests/native_engine_test.c")
    target_link_libraries(native_engine_test PRIVATE corvus_frontend)
    target_compile_options(native_engine_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_engine COMMAND native_engine_test)
    add_executable(native_frontend_test "${CORVUS_ROOT}/tests/native_frontend_test.c")
    target_link_libraries(native_frontend_test PRIVATE corvus_frontend)
    target_compile_options(native_frontend_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_frontend COMMAND native_frontend_test)
    add_executable(native_viewport_test "${CORVUS_ROOT}/tests/native_viewport_test.c")
    target_link_libraries(native_viewport_test PRIVATE corvus_frontend)
    target_compile_options(native_viewport_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME native_viewport COMMAND native_viewport_test)
    if(CORVUS_SANITIZE)
        foreach(target corvus_frontend native_engine_test native_frontend_test native_viewport_test)
            target_compile_options(${target} PRIVATE -g -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
        endforeach()
        foreach(target native_engine_test native_frontend_test native_viewport_test)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endforeach()
    endif()
endif()
